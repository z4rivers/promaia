SELECT DISTINCT 
u.page_id,
u.title,
u.created_time,
g.sender_email,
u.database_name
FROM unified_content u
LEFT JOIN gmail_content g ON u.page_id = g.page_id
WHERE (u.database_name IN ('gmail', 'trass.gmail'))
AND (u.created_time >= DATE('now', '-2 months'))
AND (
    LOWER(u.title) LIKE '%mgm%'
    OR LOWER(g.subject) LIKE '%mgm%'
    OR LOWER(g.sender_email) LIKE '%mgm%'
    OR LOWER(g.sender_name) LIKE '%mgm%'
)
ORDER BY u.created_time DESC
LIMIT 1000;
